// Load My Turfs async function loadTurfs() { try { const res = await fetch(`http://localhost:4001/turfs/owner/${user.id}`); const turfs = await res.json(); const container = document.getElementById("my-turfs"); const dashboard = document.getElementById("dashboard"); const addTurfSection = document.getElementById("add-turf-section"); if (turfs.length === 0) { container.innerHTML = "

No turfs added yet. Please add your first turf below.

"; dashboard.insertBefore(addTurfSection, dashboard.firstChild); } else { container.innerHTML = turfs.map(t => `
${t.location} — ${t.sport} — ₹${t.price}
`).join(""); dashboard.appendChild(addTurfSection); } } catch (err) { console.error("Error loading turfs:", err); document.getElementById("my-turfs").innerHTML = "

Failed to load turfs ❌

"; } } // Show Edit Form function showEditForm(id, location, district, sport, price, minTarget = 0, maxPlayers = 0) { const container = document.getElementById("my-turfs"); const formId = `editForm-${id}`; if (document.getElementById(formId)) return; container.innerHTML += `

Edit Turf: ${location}

`; } // Update Turf async function updateTurf(id) { let district = document.getElementById(`edit-district-${id}`).value.trim(); district = district.charAt(0).toUpperCase() + district.slice(1).toLowerCase(); const updated = { location: document.getElementById(`edit-location-${id}`).value, district, sport: document.getElementById(`edit-sport-${id}`).value, price: parseInt(document.getElementById(`edit-price-${id}`).value), min_players: parseInt(document.getElementById(`edit-minTarget-${id}`).value), max_stranger_players: parseInt(document.getElementById(`edit-maxPlayers-${id}`).value) }; try { await fetch(`http://localhost:4001/turfs/${id}`, { method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify(updated) }); alert("Turf updated successfully ✅"); loadTurfs(); } catch (err) { console.error("Error updating turf:", err); alert("Failed to update turf ❌"); } } // Delete Turf async function deleteTurf(id) { if (!confirm("Are you sure you want to delete this turf?")) return; try { await fetch(`http://localhost:4001/turfs/${id}`, { method: "DELETE" }); alert("Turf deleted successfully 🗑️"); loadTurfs(); } catch (err) { console.error("Error deleting turf:", err); alert("Failed to delete turf ❌"); } } // Load Bookings async function loadBookings() { try { const res = await fetch(`http://localhost:4001/owner/bookings/${user.id}`); let bookings = await res.json(); const statusFilter = document.getElementById("statusFilter").value; const modeFilter = document.getElementById("modeFilter").value; if (statusFilter) bookings = bookings.filter(b => b.status === statusFilter); if (modeFilter) bookings = bookings.filter(b => b.mode === modeFilter); bookings.sort((a, b) => new Date(a.slot_start) - new Date(b.slot_start)); const container = document.getElementById("owner-bookings"); container.innerHTML = bookings.length ? bookings.map(b => `

${b.customer_name} booked ${b.turf_name} (${b.slot_start} - ${b.slot_end})

Status: ${b.status}

`).join("") : "

No bookings yet.

"; } catch (err) { console.error("Error loading bookings:", err); } } // Update Booking Status async function updateBookingStatus(bookingId, status) { try { await fetch(`http://localhost:4001/bookings/${bookingId}/status`, { method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ status }) }); if (status === "canceled") { alert("Booking canceled. Refund will be processed 💸"); } else { alert("Booking confirmed ✅"); } loadBookings(); } catch (err) { console.error("Error updating booking status:", err); alert("Failed to update booking ❌"); } } // Pending Credits async function loadPendingCredits() { try { const res = await fetch(`http://localhost:4001/owner/pending-credits/${user.id}`); const credits = await res.json(); const container = document.getElementById("pending-credits"); container.innerHTML = credits.length ? credits.map(c => `
${c.location} — ${c.sport}
Slot: ${c.slot_start} - ${c.slot_end}
Pending Amount: ₹${c.amount}
`).join("") : "

No pending credits at the moment ✅

"; } catch (err) { console.error("Error loading pending credits:", err); document.getElementById("pending-credits").innerHTML = "

Failed to load pending credits ❌

"; } } // Filters document.getElementById("statusFilter").addEventListener("change", loadBookings); document.getElementById("modeFilter").addEventListener("change", loadBookings); // Show wallet balance in nav async function loadUserWallet() { const user = JSON.parse(localStorage.getItem("user")); try { const res = await fetch(`http://localhost:4001/wallet/${user.id}`); const data = await res.json(); document.getElementById("wallet").innerText = `Owner's Wallet: ₹${data.balance}`; } catch (err) { console.error("Error loading wallet:", err); document.getElementById("wallet").innerText = "Owner's Wallet: ₹0"; } } loadUserWallet(); // Initial load loadTurfs(); loadBookings(); loadPendingCredits(); loadOwnerWallet();
